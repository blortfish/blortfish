<script type="text/javascript">

function tagit()
{
	var meta = document.getElementsByTagName("meta");
	var page_name = meta[0].getAttribute('property');
	var nav = document.getElementById("nav");

	var nav_list = nav.getElementsByTagName("a");
	if(page_name == "home")
	{
		nav_list[0].setAttribute("id", "current");
	}
	else if(page_name == "why")
	{
		nav_list[1].setAttribute("id", "current");
	}
	else if(page_name == "how")
	{
		nav_list[2].setAttribute("id", "current");
	}
	else if(page_name == "try")
	{
		nav_list[3].setAttribute("id", "current");
	}
	else if(page_name == "company")
	{
		nav_list[4].setAttribute("id", "current");
	}
	
}

$(document).ready(tagit); 
	
</script>
